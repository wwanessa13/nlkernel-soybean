Yield

load("EZ.rda")
load("GDec.rda")

y <- readxl::read_excel("Y_CV.xlsx") %>% pull(yield)

  list(X = EZ, model = "FIXED"),
  list(V = GDec$vectors, d = GDec$values, model = "RKHS")
)

set.seed(1)

modelo <- BGLR(
  y = y,
  ETA = ETA,
  nIter = 20000,
  burnIn = 5000,
  verbose = FALSE
)

save(modelo, file = "GBLUP_Yield_G.rda")
