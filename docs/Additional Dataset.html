8

y <- read.table("fen.txt") %>% pull(V1)
n <- length(y)

set.seed(123) 
folds <- sample(rep(1:5, length.out = n))

acc_folds <- numeric(5)
predict_list <- list()

for (f in 1:5) {
  cat("\n Fold Processing", f, "\n")
  
  idx_test  <- which(folds == f)

  y_na <- y
  y_na[idx_test] <- NA
  
  ETA <- list(list(K = Kmat, model = "RKHS"))
  
  fit <- BGLR(
    y = y_na,
    ETA = ETA,
    nIter = 20000,   
    burnIn = 5000, 
    thin = 10,
    verbose = FALSE
  )
  
  yhat_test <- fit$yHat[idx_test]
  
  acc_folds[f] <- cor(yhat_test, y[idx_test], use = "complete.obs")
  
  predict_list[[f]] <- data.frame(
    Fold = f,
    Genotype = idx_test,
    Observed = y[idx_test],
    Predicted = yhat_test
  )
}

final_results <- do.call(rbind, predict_list)
final_results

results <- data.frame(
  Model = "GBLUP",
  Mean_Accuracy = mean(acc_folds),
  SD_Accuracy = sd(acc_folds)
)

results

writexl::write_xlsx(results, "8GBLUP.xlsx")

8

y <- read.table("fen.txt") %>% pull(V1)
n <- length(y)

set.seed(123)
folds <- sample(rep(1:5, length.out = n))

results <- data.frame()
predicoes  <- list()

for (name_Model in names(sg)) {
  
  cat("\nProcessando Model:", name_Model, "\n")
  
  Kmat <- sg[[name_Model]]
  
  acc_folds <- numeric(5)
  predict_list <- list()
  
  for (f in 1:5) {
    
    cat("  Fold", f, "\n")
    
    idx_test <- which(folds == f)
    
    y_na <- y
    y_na[idx_test] <- NA
    
    ETA <- list(list(K = Kmat, model = "RKHS"))
    
    fit <- BGLR(
      y = y_na,
      ETA = ETA,
      nIter = 20000,
      burnIn = 5000,
      thin = 10,
      verbose = FALSE
    )
    
    yhat_test <- fit$yHat[idx_test]
    
    acc_folds[f] <- cor(yhat_test, y[idx_test], use = "complete.obs")
    
    predict_list[[f]] <- data.frame(
      Model    = name_Model,
      Fold      = f,
      Genotype = idx_test,
      Observed = y[idx_test],
      Predicted   = yhat_test
    )
  }
  
  predicoes[[name_Model]] <- do.call(rbind, predict_list)
  
  results <- rbind(
    results,
    data.frame(
      Model         = name_Model,
      Mean_Accuracy = mean(acc_folds),
      SD_Accuracy    = sd(acc_folds)
    )
  )
}

results <- results[order(-results$Mean_Accuracy), ]

print(results)

writexl::write_xlsx(results, "8LPC.xlsx")

Bessel

sg  <- c(0.1, 0.5, 1)
ord  <- c(0, 1, 2)
dg <- c(2, 3)

hyper <- list()

for (i in sg) {
  for (j in ord) {
    for (l in dg) {
      Kmat <- kernelMatrix(besseldot(sigma = i, 
                                     order = j, 
                                     degree = l), 
                           SNPs)
      hyper[[paste0("sigma", i, "order", j, "degree", l)]] <- Kmat
    }
  }
}
